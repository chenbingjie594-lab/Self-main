nohup bash -c '
set -euo pipefail

PY=/mnt/sda/cbj/miniconda3/envs/yolo11/bin/python
REAL=/mnt/sda/cbj/dataset/Plastic_Bomo/GLRR_DWBG_40_40_s42/real_only
OOF=/mnt/sda/cbj/dataset/Plastic_Bomo/DWBG_oof_rebuild_s42
RUNS=/mnt/sda/cbj/ImageGeneration/Self-main/runs/dwbg_oof_rebuild
PROFILE=results/dwbg/dwbg_weakness_rebuild_s42.json
V1_SCORED=results/dwbg/stage2/scored_candidates_rebuild_s42.json
V1_MANIFEST=/mnt/sda/cbj/dataset/Plastic_Bomo/GLRR_DWBG_40_40_s42/dwbg_manifest.json

V2_DIR=results/dwbg/v2/fixed_s42
BANK_DIR=$V2_DIR/feature_banks
V2_SCORED=$V2_DIR/scored_candidates_v2_s42.json
V2_DATA=/mnt/sda/cbj/dataset/Plastic_Bomo/GLRR_DWBG_v2_40_40_fixed_s42

echo "[1/3] Build detector-specific real feature banks"
$PY tools/build_dwbg_real_feature_bank.py \
  --fold "$OOF/fold_0/images/train" \
         "$OOF/fold_0/labels/train" \
         "$RUNS/fold_0_s42/weights/best.pt" \
  --fold "$OOF/fold_1/images/train" \
         "$OOF/fold_1/labels/train" \
         "$RUNS/fold_1_s42/weights/best.pt" \
  --fold "$OOF/fold_2/images/train" \
         "$OOF/fold_2/labels/train" \
         "$RUNS/fold_2_s42/weights/best.pt" \
  --output_dir "$BANK_DIR" \
  --device 3 \
  --imgsz 1536 \
  --k 5 \
  --quantile 0.95

echo "[2/3] Add real-manifold and consensus scores"
$PY tools/score_dwbg_candidates_v2.py \
  --scored_candidates "$V1_SCORED" \
  --bank "$BANK_DIR/real_feature_bank_fold0.npz" \
         "$BANK_DIR/real_feature_bank_fold0_metadata.json" \
         "$RUNS/fold_0_s42/weights/best.pt" \
  --bank "$BANK_DIR/real_feature_bank_fold1.npz" \
         "$BANK_DIR/real_feature_bank_fold1_metadata.json" \
         "$RUNS/fold_1_s42/weights/best.pt" \
  --bank "$BANK_DIR/real_feature_bank_fold2.npz" \
         "$BANK_DIR/real_feature_bank_fold2_metadata.json" \
         "$RUNS/fold_2_s42/weights/best.pt" \
  --config configs/dwbg_v2.json \
  --output "$V2_SCORED" \
  --device 3 \
  --imgsz 1536

echo "[3/3] Build Real + DWBG-v2 40+40 dataset"
$PY tools/select_dwbg_candidates_v2.py \
  --real_root "$REAL" \
  --profile "$PROFILE" \
  --scored_candidates "$V2_SCORED" \
  --config configs/dwbg_v2.json \
  --v1_manifest "$V1_MANIFEST" \
  --output_root "$V2_DATA" \
  --output_dir "$V2_DIR" \
  --flash_count 40 \
  --black_count 40 \
  --seed 42

echo "DWBG-v2 build complete."
' > logs/dwbg/dwbg_v2_fixed_s42.log 2>&1 &
