cd /mnt/sda/cbj/ImageGeneration/Self-main
TASK17_TRAINONLY_RUN=$(date +%Y%m%d_%H%M%S)
mkdir -p logs/plastic_bomo_stage17a_trainonly/$TASK17_TRAINONLY_RUN

nohup env TASK17_TRAINONLY_RUN="$TASK17_TRAINONLY_RUN" CUDA_VISIBLE_DEVICES=3 PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True bash -c '
set -e
cd /mnt/sda/cbj/ImageGeneration/Self-main
TASK17_CLEAN_RESULT=results_for_gpt/plastic_bomo_stage17a_vanilla_utility_stability/trainonly_$TASK17_TRAINONLY_RUN

/mnt/sda/cbj/miniconda3/envs/yolo11/bin/python tools/prepare_plastic_bomo_stage17a_trainonly.py \
  --protocol configs/plastic_bomo_stage17a_trainonly.json \
  --frozen results_for_gpt/plastic_bomo_stage16a_r0_traceability_preflight/annotation_provenance_audit.json \
  --real_data_yaml /mnt/sda/cbj/dataset/Plastic_Bomo/DWBG_v2_final_s42/real_only.yaml \
  --crop_source results_for_gpt/plastic_bomo_stage17a_vanilla_utility_stability/historical_source_recovery/recovered_historical_crop_source.txt \
  --mask_source results_for_gpt/plastic_bomo_stage17a_vanilla_utility_stability/historical_isolation_recovery/recovered_historical_mask_source.txt \
  --dataset_root /mnt/sda/cbj/dataset/Plastic_Bomo/Stage17A_trainonly_$TASK17_TRAINONLY_RUN \
  --output "$TASK17_CLEAN_RESULT/preparation" \
  > logs/plastic_bomo_stage17a_trainonly/$TASK17_TRAINONLY_RUN/preparation.log 2>&1

/mnt/sda/cbj/miniconda3/envs/yolo11/bin/python tools/train_plastic_bomo_stage17a_trainonly.py \
  --protocol configs/plastic_bomo_stage17a_trainonly.json \
  --prepared "$TASK17_CLEAN_RESULT/preparation" \
  --frozen results_for_gpt/plastic_bomo_stage16a_r0_traceability_preflight/annotation_provenance_audit.json \
  --real_data_yaml /mnt/sda/cbj/dataset/Plastic_Bomo/DWBG_v2_final_s42/real_only.yaml \
  --base_model /mnt/sda/cbj/ImageGeneration/Self-main/sd2-community/stable-diffusion-2-inpainting \
  --original_rebuild_audit results_for_gpt/plastic_bomo_stage17a_vanilla_utility_stability/baseline_rebuild \
  --model_root /mnt/sda/cbj/ImageGeneration/Self-main/model/baseline_stage17a_trainonly_s42_noise_0.0 \
  --output "$TASK17_CLEAN_RESULT/training" \
  --logs logs/plastic_bomo_stage17a_trainonly/$TASK17_TRAINONLY_RUN/classes
' > logs/plastic_bomo_stage17a_trainonly/$TASK17_TRAINONLY_RUN/launcher.log 2>&1 &

echo $! > logs/plastic_bomo_stage17a_trainonly/$TASK17_TRAINONLY_RUN/launcher.pid
echo "本次目录：$TASK17_TRAINONLY_RUN"
# 在同一个终端查看状态：
ps -fp "$(cat logs/plastic_bomo_stage17a_trainonly/$TASK17_TRAINONLY_RUN/launcher.pid)"
